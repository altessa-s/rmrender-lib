function(rmr_target target)
    set_target_properties(${target} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF
        C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
    get_target_property(kind ${target} TYPE)
    if(NOT kind STREQUAL "EXECUTABLE")
        set_target_properties(${target} PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR})
    endif()
    target_compile_features(${target} PUBLIC c_std_23)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wshadow -Wconversion -Wsign-conversion
        -ffp-contract=off -Wstrict-prototypes -Wmissing-prototypes -Wpointer-arith -Wcast-qual)
    if(RMRENDER_WERROR)
        target_compile_options(${target} PRIVATE -Werror)
    endif()
    target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
endfunction()
